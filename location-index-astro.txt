---
import BaseLayout from '../../layouts/BaseLayout.astro';
import SiteHeader from '../../components/SiteHeader.astro';
import SiteFooter from '../../components/SiteFooter.astro';
import LocationSideNav from '../../components/LocationSideNav.astro';
import SchoolSquareCard from '../../components/SchoolSquareCard.astro';
import {
  getAgeLabel,
  getFormatLabel,
  getGenderLabel,
  getLocationDirectoryData,
  getLocationPaths,
  getMapType,
  getPhaseLabel
} from '../../lib/directory';

export async function getStaticPaths() {
  return await getLocationPaths();
}

const BATH_COORDINATE_FALLBACKS: Record<string, { lat: number; lng: number }> = {
  'king-edwards-school': { lat: 51.386488, lng: -2.343663 },
  'kingswood-school': { lat: 51.398883, lng: -2.370005 },
  'kingswood-preparatory-school': { lat: 51.397143, lng: -2.373238 },
  'prior-park-college': { lat: 51.364523, lng: -2.343082 },
  'paragon-school': { lat: 51.370494, lng: -2.355122 },
  'monkton-combe-school': { lat: 51.357305, lng: -2.326354 },
  'royal-high-school-bath-gdst': { lat: 51.397185, lng: -2.365419 },
  'bath-academy': { lat: 51.383903, lng: -2.363978 },
  'downside-school': { lat: 51.253899, lng: -2.495195 }
};

const { location: locationSlug } = Astro.params;
if (!locationSlug) throw new Error('Missing location slug');

const { location, schools, schoolCards } = await getLocationDirectoryData(locationSlug);
const fallbackMetaTitle = `${location.name} private schools | The Private School Guide`;
const fallbackMetaDescription = `Independent guide to ${location.name} private schools. Compare schools, review fees and bursaries, explore open days, and use the map to narrow your shortlist.`;
const heroImage = '/assets/img/bath/bath-location-hero.jpg';

const mapSchools = schools
  .map((school) => {
    const bathFallback = location.slug === 'bath' ? BATH_COORDINATE_FALLBACKS[school.slug] : null;
    const lat = school.latitude !== null ? Number(school.latitude) : bathFallback?.lat ?? null;
    const lng = school.longitude !== null ? Number(school.longitude) : bathFallback?.lng ?? null;

    if (!Number.isFinite(lat) || !Number.isFinite(lng)) {
      return null;
    }

    return {
      name: school.name,
      href: `/${location.slug}/schools/${school.slug}/`,
      lat,
      lng,
      type: getMapType(school.phase, school.age_max),
      note: `${getPhaseLabel(school.phase, school.age_max)} · ${getGenderLabel(school.gender)} · ${getFormatLabel(school.day_boarding)} · Ages ${getAgeLabel(school.age_min, school.age_max)}`
    };
  })
  .filter(Boolean);

const locationMapScript = `window.locationPageMapData = ${JSON.stringify(mapSchools)};`;

    </div>
  </main>
